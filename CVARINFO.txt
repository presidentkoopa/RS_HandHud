// RS_HandHUD -- readouts stuck to things that move.
//
// FOUR MOUNTS, FOUR PREFIXES, ONE SHAPE EACH:
//
//     rs_handhud_m    main wrist      rs_handhud_gm   main gun
//     rs_handhud_o    off wrist       rs_handhud_go   off gun
//
// Every prefix carries the same set. The placement half of it --
// _ofs_x/_ofs_y/_ofs_z, _yaw/_pitch/_roll, _scale, _scale_x/y/z -- is NOT
// read by ZScript at all: MODELDEF names the prefix on a PlacementCVars line
// and the RENDERER reads them, every frame, for that model. Adding a fifth
// mount means adding a fifth prefix here and a fifth MODELDEF block, and
// nothing else.

// ============================================================== the master --
user bool rs_handhud = true;

// Ignore the wrist gate and hold both wrist plates up. For placing them --
// without it they keep fading out while you are looking at them. Gun plates
// are never gated, so this does not affect them.
user bool rs_handhud_always = false;

// The status bar's own animated face on whichever plate shows vitals.
user bool rs_handhud_mugshot = true;

// Swap which hand's weapon the WRIST plates describe, for a left-hander. The
// two gun plates always describe the gun they are bolted to.
user bool rs_handhud_swap = false;

// How far off its target roll a wrist may be and still count as turned
// toward you. Degrees, shared by both wrists.
user float rs_handhud_roll_tol = 45.0;

// ============================================================== diagnostic --
// Once a second: both wrist rolls, which plates are up, and every number
// being painted.
user bool rs_handhud_debug = false;

// PAINT A KNOWN PATTERN INSTEAD OF THE READOUT -- three coloured bars and a
// digit per panel. If the plate shows this and not the readout, the fault is
// in what the readout was given, not in the canvas, the skin, the model, the
// layer or the placement. That distinction cost a day to make by hand.
user bool rs_handhud_selftest = false;

// READ A MOD'S MAGAZINE BY FIELD NAME. Empty means "try the built-in list of
// names mods actually use, then Ammo2, then a matching inventory item". Set
// it to one field name and that name is used instead, for every weapon --
// which is how a mod none of those three shapes covers gets read without a
// code change.
//
// Run `netevent rs-handhud-probe` to print every field on the weapons in your
// hands, every inventory item that could be a magazine, and which one matched.
user string rs_handhud_magfield = "";

// ================================================== main wrist -- behaviour --
user bool  rs_handhud_m_on   = true;
// 0 ammo, 1 vitals, 2 both.
user int   rs_handhud_m_role = 0;
// The wrist roll at which this plate is turned toward you.
user float rs_handhud_m_roll_gate = 90.0;

// --------------------------------------- main wrist -- placement (renderer) --
// TUNED IN A HEADSET 2026-09-04. Not a starting guess.
user float rs_handhud_m_ofs_x = 0.0;
user float rs_handhud_m_ofs_y = 23.0;
user float rs_handhud_m_ofs_z = 14.0;
user float rs_handhud_m_yaw   = 0.0;
user float rs_handhud_m_pitch = 0.0;
user float rs_handhud_m_roll  = 0.0;
user float rs_handhud_m_scale = 100.0;

// =================================================== off wrist -- behaviour --
user bool  rs_handhud_o_on   = true;
user int   rs_handhud_o_role = 1;
user float rs_handhud_o_roll_gate = -90.0;

// ---------------------------------------- off wrist -- placement (renderer) --
user float rs_handhud_o_ofs_x = 0.0;
user float rs_handhud_o_ofs_y = 23.0;
user float rs_handhud_o_ofs_z = 14.0;
user float rs_handhud_o_yaw   = 0.0;
user float rs_handhud_o_pitch = 0.0;
user float rs_handhud_o_roll  = 0.0;
user float rs_handhud_o_scale = 100.0;

// ==================================================== main gun -- behaviour --
// OFF by default: an on-gun readout has to be placed for the weapon set in
// front of you, and one that arrives already floating beside a gun reads as a
// bug rather than a feature. Turn it on, turn rs_handhud_always on, place it.
user bool  rs_handhud_gm_on   = false;
user int   rs_handhud_gm_role = 0;

// ----------------------------------------- main gun -- placement (renderer) --
user float rs_handhud_gm_ofs_x = 0.0;
user float rs_handhud_gm_ofs_y = 0.0;
user float rs_handhud_gm_ofs_z = 0.0;
user float rs_handhud_gm_yaw   = 0.0;
user float rs_handhud_gm_pitch = 0.0;
user float rs_handhud_gm_roll  = 0.0;
user float rs_handhud_gm_scale = 60.0;

// ===================================================== off gun -- behaviour --
user bool  rs_handhud_go_on   = false;
user int   rs_handhud_go_role = 0;

// ------------------------------------------ off gun -- placement (renderer) --
user float rs_handhud_go_ofs_x = 0.0;
user float rs_handhud_go_ofs_y = 0.0;
user float rs_handhud_go_ofs_z = 0.0;
user float rs_handhud_go_yaw   = 0.0;
user float rs_handhud_go_pitch = 0.0;
user float rs_handhud_go_roll  = 0.0;
user float rs_handhud_go_scale = 60.0;

// BRING THE PLATES UP WITHOUT VR. For a bench run, not for play.
//
// The handler normally waits for the engine to be driving the hands before it
// installs anything, which is correct and which also meant nothing past that
// gate could be tested without putting the headset on. Everything downstream
// -- the layers, the resolve, the paint -- works fine on a desktop boot; the
// plates just sit wherever the hands would be.
//
//   doomxr -iwad doom2.wad -file RS_VR_Unified.pk3 RS_HandHud.pk3 //          -nosound -stdout +vr_mode 0 +map map01 //          +rs_handhud_debug 1 +rs_handhud_bench 1
//
// and read [HandHUD/bench] on stdout.
user bool rs_handhud_bench = false;
