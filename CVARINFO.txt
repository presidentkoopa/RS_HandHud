// ============================================================================
// RS_HandHUD -- the HUD on your hands (zscript/handhud/rs_handhud.zs).
// ============================================================================
user bool  rs_handhud = true;

// Plate size, on top of the MODELDEF Scale.
// TUNED IN THE HEADSET 2026-09-04. These are where the plates actually sit
// and how big they actually read; they are not a starting guess.
user float rs_handhud_scale = 100.0;

// Weapon plate on the main wrist and vitals on the off forearm by default;
// on, the two swap (left-handers).
user bool  rs_handhud_swap = false;

// THE WRIST GATE. A plate fades in when that wrist is rolled to within
// rs_handhud_roll_tol degrees of its target roll, and out otherwise -- the
// way you check a watch. The engine's roll fields turn opposite to actor
// roll, so tune the targets by feel rather than by reasoning about sign.
user float rs_handhud_roll_main = 90.0;
user float rs_handhud_roll_off  = -90.0;
user float rs_handhud_roll_tol  = 45.0;

// Fade length in tics.
user float rs_handhud_fade = 6.0;

// Ignore the gate: both plates always up. For placing them.
user bool  rs_handhud_always = false;

// The status bar face on the vitals plate.
user bool  rs_handhud_mugshot = true;

// Once a second on the console: both wrist rolls, the gate verdicts, whether
// each layer is up, and the numbers being painted.
user bool  rs_handhud_debug = false;

// ---------------------------------------------------------------- placing --
// WHERE ON THE WRIST, per hand, live. These reach the plate through the
// psprite's AnchorOfs/AnchorAngles, which the engine only honours for a layer
// anchored to another -- the handler anchors each plate to that hand's palm
// bone so these work at all.
//
// Offsets are in MAP UNITS in the hand's own frame; 34 units is a metre, so
// most of the useful range is single digits. Angles are degrees.
//
// Turn rs_handhud_always on while setting these, or the plate keeps fading
// out while you are trying to look at it.
user float rs_handhud_m_ofs_x = 0.0;
user float rs_handhud_m_ofs_y = 23.0;
user float rs_handhud_m_ofs_z = 14.0;
user float rs_handhud_m_yaw   = 0.0;
user float rs_handhud_m_pitch = 0.0;
user float rs_handhud_m_roll  = 0.0;

user float rs_handhud_o_ofs_x = 0.0;
user float rs_handhud_o_ofs_y = 23.0;
user float rs_handhud_o_ofs_z = 14.0;
user float rs_handhud_o_yaw   = 0.0;
user float rs_handhud_o_pitch = 0.0;
user float rs_handhud_o_roll  = 0.0;

// READ A MOD'S MAGAZINE BY FIELD NAME. Empty means "try the built-in list of
// names mods actually use". Set it to one field name and that name is used
// instead, for every weapon -- which is how a mod the list does not cover
// gets read without a code change.
//
// Run `netevent rs-handhud-probe` to print every field on the weapons in
// your hands, then put the one that holds the count in here.
user string rs_handhud_magfield = "";

// THE PLATE'S SHAPE. 0 is the flat slab it started as; 1 folds it into a
// triptych -- the middle panel square to you, the two flanking it hinged back
// 40 degrees to follow the wrist down its sides, so the edges neither float
// off the arm nor bury themselves in it.
//
// Both meshes share a footprint and a UV layout, so switching costs nothing:
// the placement stays where it was tuned and the canvas paints the same.
user int rs_handhud_shape = 1;
