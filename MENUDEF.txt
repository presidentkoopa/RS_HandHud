// RS_HandHUD -- the menu.
//
// FOUR PAGES THAT ARE THE SAME PAGE. Every mount exposes an identical shape --
// on/off, what it says, and seven placement sliders -- because a mount IS a
// prefix plus a layer. A fifth readout is a fifth copy of one of these blocks
// with the prefix swapped, and nothing else.
//
// THE PLACEMENT SLIDERS DO NOT GO THROUGH ZSCRIPT. MODELDEF names the prefix on
// a PlacementCVars line and the RENDERER reads <prefix>_ofs_x and friends every
// frame, so dragging a slider moves the plate in the headset immediately, with
// the menu still open and no tic in between. That is the whole reason these are
// cvars rather than script state.
//
// Turn on "Hold plates up" before placing anything. Without it a wrist plate
// fades out the moment you turn your wrist toward the menu instead of toward
// your face -- which is exactly when you are trying to look at it.

// Entry point. This hangs off the engine's own Options menu so RS_HandHUD is
// reachable loaded on its own. When it folds into RS_VR_Unified these two lines
// become AddOptionMenu "RS_VRUnifiedOptions" and it moves under that hub.
AddOptionMenu "OptionsMenu"
{
	Submenu "Hand HUD", "RS_HandHUDOptions"
}
AddOptionMenu "OptionsMenuSimple"
{
	Submenu "Hand HUD", "RS_HandHUDOptions"
}

OptionValue "RS_HandHUDRenderer"
{
	0, "Canvas plates"
	1, "Layered digits"
}

OptionValue "RS_HandHUDRole"
{
	0, "Ammo"
	1, "Vitals"
	2, "Both"
}

// =========================================================================
OptionMenu "RS_HandHUDOptions"
{
	Title "HAND HUD"

	StaticText ""
	Option "Hand HUD", "rs_handhud", "OnOff"

	// Same numbers either way; only the delivery differs. Layered digits are
	// Ermac's mechanism -- a psprite layer per digit, nothing painted, so
	// nothing can come up blank. Canvas plates can draw bars, icons and a
	// mugshot anywhere on a mesh folded to the wrist, and have more ways to
	// fail. Both stay.
	Option "Draw with", "rs_handhud_renderer", "RS_HandHUDRenderer"
	Submenu "Layered digits placement", "RS_HandHUDLayered"

	StaticText " "
	StaticText "Readouts", gold
	Submenu "Main wrist", "RS_HandHUDMainWrist"
	Submenu "Off wrist",  "RS_HandHUDOffWrist"
	Submenu "Main gun",   "RS_HandHUDMainGun"
	Submenu "Off gun",    "RS_HandHUDOffGun"

	StaticText " "
	StaticText "Behaviour", gold
	// On while placing. The wrist gate is the first thing that gets in the way
	// of tuning the thing the gate applies to.
	Option "Hold plates up (for placing)", "rs_handhud_always",   "OnOff"
	Option "Mugshot",                      "rs_handhud_mugshot",  "OnOff"
	// Makes both wrists describe the other hand's weapon. For a left-hander.
	Option "Swap hands (left-handed)",     "rs_handhud_swap",     "OnOff"
	Slider "Wrist turn tolerance",         "rs_handhud_roll_tol", 5, 180, 5, 0

	StaticText " "
	StaticText "Diagnostics", gold
	// Three coloured bars and a digit per panel. If THIS shows and the readout
	// does not, the fault is in the numbers, not in the canvas, the skin, the
	// model, the layer or the placement -- which is the one question worth
	// being able to answer without guessing at it.
	Option "Self-test pattern", "rs_handhud_selftest", "OnOff"
	Option "Debug printout",    "rs_handhud_debug",    "OnOff"
	// Prints every field on the weapons in your hands, every inventory item
	// that could be a magazine, and which one the reader matched -- so a mod
	// that keeps its clip somewhere new can be wired up by name.
	SafeCommand "Probe held weapons", "netevent rs-handhud-probe"
	StaticText "then: rs_handhud_magfield <name>"

	StaticText " "
	SafeCommand "Reset everything", "resetcvar rs_handhud rs_handhud_always rs_handhud_mugshot rs_handhud_swap rs_handhud_roll_tol rs_handhud_debug rs_handhud_selftest rs_handhud_magfield"
}
